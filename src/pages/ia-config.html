// =================================================== // 🚀 ia-config.js — versión completa optimizada (producción) // =================================================== console.log('🚀 Iniciando ia-config.js (producción)'); // ---------------------------- // 🧩 VARIABLES GLOBALES // ---------------------------- let productos = []; let iaConfig = []; // ---------------------------- // 📦 CARGA DE PRODUCTOS E IA CONFIG // ---------------------------- async function cargarDatos() { try { // Cargar catálogo real const resProd = await fetch('productos.json'); productos = await resProd.json(); // Cargar configuración de la IA const resIa = await fetch('ia-config.json'); iaConfig = await resIa.json(); console.log(`✅ Productos cargados: ${productos.length}`); console.log('📦 Ejemplo de productos cargados:', productos.slice(0, 3)); console.log('✅ IA Config cargada y sincronizada con productos reales:', iaConfig); } catch (err) { console.error('❌ Error cargando datos:', err); } } // ---------------------------- // 🧭 INICIALIZACIÓN PRINCIPAL // ---------------------------- document.addEventListener('DOMContentLoaded', async () => { console.log('📥 Iniciando carga de datos...'); await cargarDatos(); inicializarBuscador(); inicializarControlesIA(); console.log('✅ Todo listo para interactuar con el panel.'); }); // ---------------------------- // 🎛️ INICIALIZAR CONTROLES IA // ---------------------------- function inicializarControlesIA() { const guardarBtn = document.getElementById('guardarIAConfig'); if (guardarBtn) { guardarBtn.addEventListener('click', guardarConfiguracionIA); } console.log('🧠 Controles IA inicializados.'); } // ---------------------------- // 💾 GUARDAR CONFIGURACIÓN DE IA // ---------------------------- function guardarConfiguracionIA() { try { const data = iaConfig.map(item => ({ id: item.id, activo: item.activo || false, prioridad: item.prioridad || 0 })); console.log('💾 Configuración IA guardada:', data); // Simular guardado en servidor alert('✅ Configuración de IA actualizada correctamente'); } catch (err) { console.error('❌ Error guardando IA config:', err); } } // =================================================== // 🔍 BÚSQUEDA REACTIVA DE PRODUCTOS // =================================================== let debounceTimer = null; function inicializarBuscador() { const inputBusqueda = document.getElementById('searchProductos'); const resultadosContainer = document.getElementById('searchResults'); if (!inputBusqueda || !resultadosContainer) { console.warn('⚠️ No se encontraron elementos de búsqueda.'); return; } console.log('✅ Input y contenedor de resultados listos'); // Escuchar entrada de texto inputBusqueda.addEventListener('input', (e) => { const termino = e.target.value.trim(); if (!termino) { ocultarResultados(); return; } clearTimeout(debounceTimer); debounceTimer = setTimeout(() => buscarProductos(termino), 300); }); // Ocultar al hacer clic fuera document.addEventListener('click', (e) => { if (!resultadosContainer.contains(e.target) && e.target !== inputBusqueda) { ocultarResultados(); } }); } // ---------------------------- // 🔎 FUNCIÓN DE BÚSQUEDA AVANZADA // ---------------------------- function buscarProductos(termino) { const resultadosContainer = document.getElementById('searchResults'); if (!resultadosContainer) return; termino = termino.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, ''); console.log(`🔍 Buscando: ${termino}`); const resultados = productos.filter(p => { const campos = [ p.nombre, p.codigo, p.marca, p.descripcion, p.categoria ] .filter(Boolean) .map(v => v.toString().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '')); return campos.some(campo => campo.includes(termino)); }); console.log(`📊 Resultados encontrados: ${resultados.length}`); resultadosContainer.innerHTML = ''; if (resultados.length === 0) { ocultarResultados(); return; } resultados.slice(0, 10).forEach(prod => { const div = document.createElement('div'); div.className = 'producto-item'; div.innerHTML = ` ${prod.nombre}
Código: ${prod.codigo || '—'} — ${prod.marca || ''} `; div.addEventListener('click', () => { document.getElementById('searchProductos').value = prod.nombre; ocultarResultados(); console.log('🟢 Producto seleccionado:', prod); }); resultadosContainer.appendChild(div); }); mostrarResultados(); } // ---------------------------- // 🎨 MOSTRAR / OCULTAR RESULTADOS // ---------------------------- function mostrarResultados() { const cont = document.getElementById('searchResults'); cont.classList.add('show'); cont.classList.remove('hide'); cont.style.display = 'block'; } function ocultarResultados() { const cont = document.getElementById('searchResults'); cont.classList.add('hide'); cont.classList.remove('show'); setTimeout(() => (cont.style.display = 'none'), 200); }
